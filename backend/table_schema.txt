users
-----
id (PK)
uuid (unique)
first_name
last_name
email (unique, indexed)
username (unique, indexed)
hashed_password
phone

is_active (boolean default true)
is_verified (boolean default false)

last_login
created_at
updated_at

-- Flexible extension
extra_data JSONB   -- for future fields

----------------------------------------------------------------------------------------------------

roles
-----
id (PK)
name (unique)        -- admin, teacher, student
description
is_system (boolean)  -- protect core roles

----------------------------------------------------------------------------------------------------

user_roles
----------
user_id (FK -> users.id)
role_id (FK -> roles.id)
PRIMARY KEY (user_id, role_id)

----------------------------------------------------------------------------------------------------

permissions
-----------
id (PK)
name (unique)        -- create_student
module               -- student, exam
action               -- create, read, update, delete

----------------------------------------------------------------------------------------------------

role_permissions
----------------
role_id (FK)
permission_id (FK)
PRIMARY KEY (role_id, permission_id)

----------------------------------------------------------------------------------------------------

user_permissions
----------------
user_id (FK)
permission_id (FK)
is_allowed (boolean)
PRIMARY KEY (user_id, permission_id)

----------------------------------------------------------------------------------------------------
